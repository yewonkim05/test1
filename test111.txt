<script>
  // 각 방의 Zoom 링크를 여기에 입력하세요.
  const zoomRooms = {
    1: "https://zoom.us/j/1111111111", // 1번방
    2: "https://zoom.us/j/2222222222", // 2번방
    3: "https://zoom.us/j/3333333333", // 3번방
    4: "https://zoom.us/j/4444444444", // 4번방
  };

  function showRooms() {
    document.getElementById('start').style.display = 'none';
    document.getElementById('rooms').style.display = 'block';
    document.getElementById('title').textContent = "회의실을 선택하세요";
  }

  function joinRoom(num) {
    const url = zoomRooms[num];
    if (url) window.open(url, "_blank", "noopener,noreferrer");
    else alert("아직 설정되지 않은 방입니다.");
  }
</script>
